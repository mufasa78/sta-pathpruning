```python
class FeatureExtractor:
    """Extract 21-dim features per candidate node"""
    def extract(self, node, paths):
        # 7 path features + 14 node features
        return np.concatenate([
            self.path_features(paths),  # 7-dim
            self.node_features(node)     # 14-dim
        ])

# candidate_generator.py
class CandidateGenerator:
    """Stage 1: Generate ≤30 candidate nodes"""
    def generate(self, endpoint_graph):
        nodes = endpoint_graph.get_middle_region_nodes()
        return nodes[:30]  # Max 30 candidates

# anchor_predictor.py
class AnchorPredictor:
    """Stage 2: Predict anchor from candidates"""
    def __init__(self, model_type='rf'):
        self.model = RandomForestClassifier(n_estimators=1000)
    
    def predict(self, candidates, endpoint_graph):
        features = [self.extract_features(c) for c in candidates]
        probs = self.model.predict_proba(features)
        return candidates[np.argmax(probs)]
```

### Week 4: Pipeline Integration

```python
# pipeline.py - Streamlined execution
class Pipeline:
    def __init__(self):
        self.candidate_gen = CandidateGenerator()
        self.anchor_pred = AnchorPredictor()
    
    def process_endpoint(self, endpoint_graph):
        # Stage 1: Generate candidates
        candidates = self.candidate_gen.generate(endpoint_graph)
        
        # Stage 2: Predict anchor
        anchor = self.anchor_pred.predict(candidates, endpoint_graph)
        
        # Stage 3: Prune and get worst paths
        pruned_paths = [p for p in endpoint_graph.paths 
                       if anchor in p.nodes]
        
        # Return top 10 worst paths
        return sorted(pruned_paths, key=lambda p: p.slack)[:10]
```
